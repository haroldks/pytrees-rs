11/// Why a search could not produce a tree.
12#[derive(Debug, Clone, Copy, Serialize, Deserialize)]
13pub enum FitError {
14    /// The depth is not supported by this search.
15    InvalidDepth(usize),
16    /// The minimum support is invalid.
17    InvalidMinSupport(usize),
18    /// No feature can split the instances.
19    EmptyCandidates,
20    /// The search failed.
21    AlgorithmError,
22    /// Too few instances to learn from.
23    InsufficientData,
24    /// No split beats a single leaf.
25    EmptyTree,
26}
27
28impl fmt::Display for FitError {
29    fn fmt(&self, f: &mut Formatter<'_>) -> fmt::Result {
30        match self {
31            FitError::InvalidDepth(depth) => write!(f, "Invalid depth: {}", depth),
32            FitError::InvalidMinSupport(support) => {
33                write!(f, "Invalid minimum support threshold: {}", support)
34            }
35            FitError::EmptyCandidates => write!(f, "Empty candidates list found"),
36            FitError::AlgorithmError => write!(f, "Error in algorithm execution"),
37            FitError::InsufficientData => write!(f, "Insufficient data for training"),
38            FitError::EmptyTree => write!(f, "Search produced an empty tree"),
39        }
40    }
41}
42
43impl std::error::Error for FitError {}

45/// Counters collected during a search.
46#[derive(Debug, Default, Clone, Copy, Serialize, Deserialize)]
47pub struct SearchStatistics {
48    /// Number of cached subproblems.
49    pub cache_size: usize,
50    /// Subproblems found in the cache.
51    pub cache_hits: usize,
52    /// Number of passes (the first one included).
53    pub restarts: usize,
54    /// Second branches skipped because the first one used up the bound.
55    pub sibling_pruning: usize,
56    /// Number of nodes visited.
57    pub search_space_size: usize,
58    /// Training error of the best tree.
59    pub tree_error: f64,
60    /// Search time in seconds.
61    pub duration: f64,
62    /// Number of features.
63    pub num_attributes: usize,
64    /// Number of training instances.
65    pub num_samples: usize,
66}
67
68impl SearchStatistics {
69    pub fn increment_search_space(&mut self) {
70        self.search_space_size += 1;
71    }
72    pub fn increment_cache_hits(&mut self) {
73        self.cache_hits += 1;
74    }
75
76    pub fn increment_sibling_pruning(&mut self) {
77        self.sibling_pruning += 1;
78    }
79
80    pub fn increment_restarts(&mut self) {
81        self.restarts += 1;
82    }
83    /// Number of passes so far.
84    pub fn restarts(&self) -> usize {
85        self.restarts
86    }
87}

160/// Heuristic used to order the features.
161#[derive(Debug, Clone, Copy, Serialize, Deserialize, Eq, PartialEq)]
162#[cfg_attr(feature = "cli", derive(clap::ValueEnum))]
163pub enum SearchHeuristic {
164    /// Keep the features in their original order.
165    NoHeuristic,
166    /// Lowest weighted Gini impurity first.
167    GiniIndex,
168    /// Highest information gain first.
169    InformationGain,
170    /// Lowest weighted entropy of the children first.
171    WeightedEntropy,
172}
173
174impl From<SearchHeuristic> for Box<dyn Heuristic> {
175    fn from(value: SearchHeuristic) -> Self {
176        match value {
177            SearchHeuristic::InformationGain => Box::<InformationGain>::default(),
178            SearchHeuristic::WeightedEntropy => Box::<WeightedEntropy>::default(),
179            SearchHeuristic::GiniIndex => Box::<GiniIndex>::default(),
180            SearchHeuristic::NoHeuristic => Box::<NoHeuristic>::default(),
181        }
182    }
183}
