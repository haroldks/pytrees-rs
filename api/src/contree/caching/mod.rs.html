1//! The cache of solved subproblems.
2//!
3//! A subproblem is "the best tree of depth `d` over this set of instances", so
4//! it is keyed by `(remaining depth, the set)`. The set is a [`Bitset`] over
5//! instance ids, and its population count is pulled out as a second array
6//! index so a lookup never hashes a bitset against entries that could not
7//! possibly match.
8//!
9//! Entries live in one arena and refer to each other by index, which is what
10//! lets the solution tree be rebuilt afterwards by walking from the root.
11
12use crate::bitsets::Bitset;
13use crate::tree::Tree;
14use rustc_hash::FxHashMap;
15use serde::{Deserialize, Serialize};
16
17/// One solved (or partly solved) subproblem.
18#[derive(Copy, Clone, Serialize, Deserialize, Debug)]
19pub struct Entry {
20    /// The feature this node tests, or `usize::MAX` for a leaf.
21    pub feature: usize,
22    /// The threshold, or `INFINITY` for a leaf.
23    pub split: f64,
24    /// Misclassifications of the best subtree found so far. This is an upper
25    /// bound on the subproblem's optimum, equal to it when `is_optimal`.
26    pub error: usize,
27    /// The majority class, used when this entry is a leaf.
28    pub label: usize,
29
30    /// A proven *lower* bound on this subproblem's optimum.
31    ///
32    /// Equal to `error` when the search exhausted the space, and to the upper
33    /// bound it was cut off at otherwise. Reasoning about what a subtree
34    /// cannot achieve must use this field, not `error`.
35    pub lower_bound: usize,
36    /// Whether `error` describes a tree that was actually found, rather than a
37    /// subproblem the bound cut short before one was.
38    pub is_valid: bool,
39
40    /// Whether the best tree for this subproblem is a single leaf.
41    pub is_leaf: bool,
42    /// Whether `error` is proven optimal for this subproblem rather than just
43    /// the best seen so far. Only such an entry can replace a new search.
44    pub is_optimal: bool,
45
46    /// Distance from the root, not remaining depth.
47    pub depth: usize,
48    /// Arena index of the left child. `0` means "no child": index 0 is the
49    /// root, which is nobody's child.
50    pub left: usize,
51    /// Arena index of the right child, with the same convention as `left`.
52    pub right: usize,
53    /// Set when the depth-2 solver produced a whole subtree for this node
54    /// instead of cache entries; indexes `Cache::trees`.
55    pub tree_idx: Option<usize>,
56    /// The budgets of the last anytime search of this subproblem. Only
57    /// `ConTreeLds` sets it; see [`SearchedUnder`].
58    pub searched_under: Option<SearchedUnder>,
59}
60
61/// The budget and bound a subproblem was last searched under, and whether the
62/// budget cut that search short.
63///
64/// A result obtained under a budget is at least as good as anything a smaller
65/// budget could find, so a revisit under a budget and bound no larger than
66/// these can reuse it. A search the budget did not cut short proved its lower
67/// bound, which settles any later revisit whose bound is no higher.
68#[derive(Copy, Clone, Serialize, Deserialize, Debug, PartialEq, Eq)]
69pub struct SearchedUnder {
70    /// Discrepancy still available below this node.
71    pub discrepancy: usize,
72    /// Best-ranked splits each node could try; `usize::MAX` when the split
73    /// budget did not apply.
74    pub split_budget: usize,
75    /// The upper bound given by the parent: no tree at or above it was useful.
76    pub upper_bound: usize,
77    /// The discrepancy or split budget cut the search short.
78    pub truncated: bool,
79}
80
81impl SearchedUnder {
82    /// Whether this search's result stands in for a search under `other`.
83    ///
84    /// It must have had at least `other`'s discrepancy and split budget. The
85    /// bound matters only when this search came back empty-handed
86    /// (`found_a_tree` false): pruning against a bound discards only what
87    /// could not beat it, so a tree found under one bound is the best the
88    /// budget allows under any looser bound too.
89    pub fn covers(&self, other: &SearchedUnder, found_a_tree: bool) -> bool {
90        other.discrepancy <= self.discrepancy
91            && other.split_budget <= self.split_budget
92            && (found_a_tree || other.upper_bound <= self.upper_bound)
93    }
94}
95
96impl Entry {
97    /// Records what the search actually proved about this subproblem.
98    ///
99    /// When the best tree found is worse than the upper bound the search ran
100    /// under, the search only established that the bound cannot be met here:
101    /// the bound becomes the lower bound, and the entry no longer counts as a
102    /// usable solution.
103    pub fn finalize_lower_bound(&mut self, upper_bound: usize) {
104        if self.error > upper_bound {
105            self.lower_bound = upper_bound;
106            self.is_valid = false;
107        } else {
108            self.lower_bound = self.error;
109            self.is_valid = true;
110        }
111    }
112
113    /// Marks the entry as an exact, reusable answer.
114    pub fn mark_exact(&mut self) {
115        self.lower_bound = self.error;
116        self.is_valid = true;
117        self.is_optimal = true;
118    }
119}
120
121impl Default for Entry {
122    fn default() -> Self {
123        Self {
124            feature: usize::MAX,
125            split: f64::INFINITY,
126            error: usize::MAX,
127            label: usize::MAX,
128            lower_bound: 0,
129            is_valid: false,
130            is_leaf: false,
131            is_optimal: false,
132            depth: 0,
133            left: 0,
134            right: 0,
135            tree_idx: None,
136            searched_under: None,
137        }
138    }
139}

141/// Arena of [`Entry`] values indexed by `(depth, population count, bitset)`,
142/// plus the subtrees produced by the depth-2 solver.
143#[derive(Debug)]
144pub struct Cache {
145    arena: Vec<Entry>,
146    trees: Vec<Tree>,
147    map: Vec<Vec<FxHashMap<Bitset, usize>>>,
148    root_index: usize,
149}
150
151impl Default for Cache {
152    fn default() -> Self {
153        Self::new(0, 0)
154    }
155}
156
157impl Cache {
158    /// Creates an empty cache for trees of at most `depth` levels.
159    ///
160    /// Buckets are allocated lazily, for the population counts each depth
161    /// actually sees.
162    pub fn new(depth: usize, _num_samples: usize) -> Self {
163        Self {
164            arena: Vec::new(),
165            trees: vec![],
166            map: vec![Vec::new(); depth + 1],
167            root_index: 0,
168        }
169    }
170
171    /// The bucket for one `(depth, count)` pair, grown on demand.
172    fn bucket(&mut self, depth: usize, count: usize) -> &mut FxHashMap<Bitset, usize> {
173        let row = &mut self.map[depth];
174        if row.len() <= count {
175            row.resize_with(count + 1, FxHashMap::default);
176        }
177        &mut row[count]
178    }
179
180    /// The root entry, once [`Self::init`] has been called.
181    pub fn root(&self) -> Option<&Entry> {
182        self.arena.get(self.root_index)
183    }
184
185    /// Arena index of the root entry.
186    pub fn root_index(&self) -> usize {
187        self.root_index
188    }
189
190    /// Creates the root entry and returns its index. The cache must be empty.
191    pub fn init(&mut self) -> usize {
192        debug_assert!(self.arena.is_empty(), "Cache must be empty to init");
193        self.arena.push(Entry::default());
194        self.root_index
195    }
196
197    /// Looks the subproblem up, inserting a fresh entry if it is new.
198    ///
199    /// Returns `(is_new, index into the arena)`.
200    pub fn insert(&mut self, bitset: &Bitset, depth: usize) -> (bool, usize) {
201        #[cfg(feature = "profiling")]
202        coz::scope!("insert in cache");
203        let count = bitset.saved_count();
204
205        if let Some(&index) = self.bucket(depth, count).get(bitset) {
206            return (false, index);
207        }
208
209        let index = self.arena.len();
210        self.arena.push(Entry::default());
211        self.bucket(depth, count).insert(bitset.clone(), index);
212        (true, index)
213    }
214
215    /// The entry at `index`.
216    pub fn get(&self, index: usize) -> Option<&Entry> {
217        self.arena.get(index)
218    }
219
220    /// The entry at `index`, mutably.
221    pub fn get_mut(&mut self, index: usize) -> Option<&mut Entry> {
222        self.arena.get_mut(index)
223    }
224
225    /// Number of entries in the cache.
226    pub fn len(&self) -> usize {
227        self.arena.len()
228    }
229
230    /// Stores a subtree built by the depth-2 solver and returns its index.
231    pub fn insert_tree(&mut self, tree: Tree) -> usize {
232        let len = self.trees.len();
233        self.trees.push(tree);
234        len
235    }
236
237    /// The subtree stored at `tree_idx`.
238    pub fn get_tree(&self, tree_idx: usize) -> Option<&Tree> {
239        self.trees.get(tree_idx)
240    }
241
242    /// Arena indices of the left and right children of entry `index`.
243    ///
244    /// # Panics
245    /// If `index` is out of bounds.
246    pub fn get_children(&self, index: usize) -> [usize; 2] {
247        assert!(index < self.arena.len());
248        [self.arena[index].left, self.arena[index].right]
249    }
250}
