11/// The best depth-1 tree found so far for one side of a depth-2 root split,
12/// and the scan state used to find it.
13///
14/// With misclassification the scan compares scores, the number of correctly
15/// classified instances, so higher is better. With a custom loss it compares
16/// costs.
17struct SubtreeLeafScores {
18    // Best split found so far.
19    classification_score: usize,
20    best_cost: Cost,
21    best_feature_index: usize,
22    best_threshold: f64,
23    best_left_label: Option<usize>,
24    best_right_label: Option<usize>,
25    best_left_error: Cost,
26    best_right_error: Cost,
27
28    // Scan state for the feature being processed.
29    previous_value: f64,
30    previous_unique_value_index: Option<usize>,
31    /// No split on this feature can improve the score any more.
32    is_zero: bool,
33    /// Instances that have to move to the left before a threshold can beat
34    /// the score: moving an instance changes the score by at most 1, so the
35    /// thresholds before that are skipped.
36    can_skip: Cost,
37    current_element_count: usize,
38    /// Costs of both leaves at the threshold evaluated last, and the number
39    /// of instances then left of it. Used to check a custom loss under
40    /// forced skipping.
41    last_evaluated: Option<(Cost, Cost, usize)>,
42    /// Scratch space for the class counts right of the threshold.
43    remaining_frequency: Vec<usize>,
44
45    // The side itself.
46    size: usize,
47    min_sup: usize,
48    max_label_frequency: usize,
49    max_label: usize,
50    /// Cost and label of the side as a single leaf.
51    leaf_cost: Cost,
52    leaf_label: usize,
53    label_frequency: Vec<usize>,
54    current_label_frequency: Vec<usize>,
55}
56
57impl SubtreeLeafScores {
58    fn new(size: usize, class_number: usize, min_sup: usize) -> Self {
59        Self {
60            classification_score: 0,
61            best_cost: Cost::INFINITY,
62            best_feature_index: 0,
63            best_threshold: -1.0,
64            best_left_label: None,
65            best_right_label: None,
66            best_left_error: 0.0,
67            best_right_error: 0.0,
68            previous_value: 0.0,
69            previous_unique_value_index: None,
70            is_zero: false,
71            can_skip: 0.0,
72            current_element_count: 0,
73            last_evaluated: None,
74            remaining_frequency: vec![0; class_number],
75            size,
76            min_sup: min_sup.max(1),
77            max_label_frequency: 0,
78            max_label: 0,
79            leaf_cost: Cost::INFINITY,
80            leaf_label: 0,
81            label_frequency: vec![0; class_number],
82            current_label_frequency: vec![0; class_number],
83        }
84    }
85
86    fn reset_label_frequency(&mut self) {
87        self.current_label_frequency.fill(0);
88        self.previous_value = 0.0;
89        self.previous_unique_value_index = None;
90        self.is_zero = false;
91        self.can_skip = 0.0;
92        self.current_element_count = 0;
93        self.last_evaluated = None;
94    }
95
96    /// Instances still to scan.
97    fn remaining(&self) -> usize {
98        self.size - self.current_element_count
99    }
100
101    /// Whether no threshold still to come can beat the best one: each needs
102    /// `can_skip` more instances moved to the left, and fewer remain.
103    fn cannot_improve(&self) -> bool {
104        self.can_skip >= self.remaining() as Cost
105    }
106
107    /// Adds the instance to the left of the thresholds still to come.
108    fn move_left(&mut self, data_point: &DataPoint) {
109        self.current_element_count += 1;
110        self.current_label_frequency[data_point.label() as usize] += 1;
111        self.previous_value = data_point.value();
112        self.previous_unique_value_index = Some(data_point.unique_value_id());
113    }
114
115    /// Instances of classes other than `label` among the scanned instances
116    /// (`left`) or the instances still to scan (`!left`): the error of a
117    /// leaf predicting `label` there.
118    fn other_classes(&self, label: usize, left: bool) -> Cost {
119        let others: usize = (0..self.label_frequency.len())
120            .filter(|&l| l != label)
121            .map(|l| {
122                if left {
123                    self.current_label_frequency[l]
124                } else {
125                    self.label_frequency[l] - self.current_label_frequency[l]
126                }
127            })
128            .sum();
129        others as Cost
130    }
131}

133/// Why the best labels of a side are set when they are read: its score starts
134/// at the leaf's score and only rises when a split also sets the labels, and
135/// they are read only when the score is above the leaf's.
136const LABELS_SET: &str = "the labels are set whenever the score beats the leaf";

138/// The loss of a depth-2 search and what the search reports about it.
139pub(crate) struct Depth2Loss<'a> {
140    pub loss: &'a Loss,
141    /// Whether thresholds are skipped; see [`Skipping`](crate::common::Skipping).
142    pub skipping: bool,
143    /// The first negative or NaN cost the loss returned.
144    pub invalid: &'a mut Option<Cost>,
145    /// Moves that broke the skipping assumption; added to the statistics.
146    violations: usize,
147}
148
149impl<'a> Depth2Loss<'a> {
150    pub fn new(loss: &'a Loss, skipping: bool, invalid: &'a mut Option<Cost>) -> Self {
151        Self {
152            loss,
153            skipping,
154            invalid,
155            violations: 0,
156        }
157    }
158
159    fn is_custom(&self) -> bool {
160        !self.loss.is_misclassification()
161    }
162
163    /// The `(cost, label)` of a leaf with these class counts. A loss that
164    /// reads instance ids never reaches the depth-2 solver.
165    fn leaf(&mut self, class_counts: &[usize]) -> (Cost, usize) {
166        checked_leaf(
167            self.loss,
168            class_counts,
169            || unreachable!("the depth-2 solver is off for losses that read instance ids"),
170            self.invalid,
171        )
172    }
173}

175/// Specialised solver for trees of depth at most 2.
176///
177/// For each root threshold it finds the best depth-1 tree on both sides in a
178/// single scan per feature, reusing the class counts of the scan, instead of
179/// recursing.
180#[derive(Default)]
181pub struct ConTreeDepth2;
182
183impl ConTreeDepth2 {
184    /// Finds the best depth-2 tree on `view` that beats `entry.error` and
185    /// `upper_bound`, updating `entry` when it does. Whether the result is
186    /// exact depends on `upper_bound`, so the caller decides, through
187    /// `Entry::finalize_lower_bound`.
188    pub fn fit(
189        &self,
190        view: &DataView<'_>,
191        search_config: &SearchConfig,
192        entry: &mut Entry,
193        upper_bound: Cost,
194        stats: &mut Statistics,
195        costs: &mut Depth2Loss<'_>,
196    ) -> Tree {
197        debug_assert!(search_config.max_depth <= 2, "Depth should be at most 2");
198
199        let num_features = view.get_feature_number();
200        let mut tree = Tree::empty_tree(2);
201        tree.update_root().map(|updater| {
202            updater
203                .error(entry.error)
204                .feature(entry.feature)
205                .label(entry.label)
206        });
207
208        for feature in 0..num_features {
209            self.expand_feature_subtree(
210                view,
211                feature,
212                search_config,
213                entry,
214                entry.error.min(upper_bound),
215                &mut tree,
216                stats,
217                costs,
218            );
219            if entry.error <= search_config.max_gap {
220                break;
221            }
222        }
223        stats.skipping_violations += std::mem::take(&mut costs.violations);
224        tree.normalize_leaves();
225        tree
226    }

228    #[allow(clippy::too_many_arguments)]
229    fn expand_feature_subtree(
230        &self,
231        view: &DataView<'_>,
232        feature: usize,
233        config: &SearchConfig,
234        entry: &mut Entry,
235        // Splits scoring at or above this bound are of no use to the caller.
236        upper_bound: Cost,
237        tree: &mut Tree,
238        stats: &mut Statistics,
239        costs: &mut Depth2Loss<'_>,
240    ) {
241        let feature_column = view.get_sorted_feature(feature);
242        let feature_column_ids = view.get_feature_indices(feature);
243        let possible_split_indices = view.get_possible_split_indices(feature);
244
245        if possible_split_indices.is_empty() {
246            return;
247        }
248
249        // Only consider root splits that leave `min_sup` instances per side.
250        let feasible = support_feasible_splits(
251            possible_split_indices,
252            view.get_dataset_size(),
253            config.min_sup,
254        );
255        if feasible.is_empty() {
256            return;
257        }
258
259        let mut pruner = IntervalsPruner::new(
260            possible_split_indices,
261            config.max_gap,
262            config.min_sup,
263            costs.skipping,
264        );
265
266        let mut queue = VecDeque::new();
267        let init_bound = Bound::new(feasible.start, feasible.end - 1, None, None);
268        queue.push_back(init_bound);
269
270        while let Some(mut current_bound) = queue.pop_front() {
271            if pruner.subinterval_pruning(&current_bound, entry.error.min(upper_bound)) {
272                continue;
273            }
274
275            pruner.interval_shrinking(&mut current_bound, entry.error.min(upper_bound));
276            if !current_bound.is_valid() {
277                continue;
278            }
279
280            let mid = (current_bound.left_bound + current_bound.right_bound) / 2;
281            let split_point = possible_split_indices[mid];
282
283            let threshold_value = if mid > 0 {
284                let previous = feature_column_ids[possible_split_indices[mid - 1]];
285                let point = feature_column_ids[split_point];
286                threshold_between(
287                    feature_column[previous].value(),
288                    feature_column[point].value(),
289                )
290            } else {
291                let point = feature_column_ids[split_point];
292                threshold_between(
293                    feature_column[feature_column_ids[0]].value(),
294                    feature_column[point].value(),
295                )
296            };
297
298            let mut left_tree = Tree::empty_tree(1);
299            let mut right_tree = Tree::empty_tree(1);
300            self.get_leaves_score(
301                view,
302                feature,
303                split_point,
304                threshold_value,
305                &mut left_tree,
306                &mut right_tree,
307                entry.error,
308                config.min_sup,
309                costs,
310            );
311            stats.specialized_solver_call += 1;
312            let current_best_error = left_tree.root_error() + right_tree.root_error();
313            if current_best_error < entry.error {
314                entry.error = current_best_error;
315                tree.update_root().map(|updater| {
316                    updater
317                        .error(entry.error)
318                        .feature(feature)
319                        .split(threshold_value)
320                });
321
322                entry.feature = tree.root_feature().unwrap_or(usize::MAX);
323                entry.split = tree.root_split().unwrap_or(f64::INFINITY);
324
325                let (left, right) = tree.node_children(tree.get_root_index());
326                tree.update_subtree(left, &left_tree, left_tree.get_root_index());
327
328                tree.update_subtree(right, &right_tree, right_tree.get_root_index());
329
330                if current_best_error == 0.0 {
331                    return;
332                }
333            }
334
335            pruner.add_result(
336                mid,
337                Some(left_tree.root_error()),
338                Some(right_tree.root_error()),
339            );
340            if current_bound.left_bound == current_bound.right_bound {
341                continue;
342            }
343
344            let score_difference = (left_tree.root_error() + right_tree.root_error()
345                - entry.error.min(upper_bound))
346            .max(0.0);
347            let (left_bound, right_bound) = pruner.neighbourhood_pruning(
348                score_difference,
349                current_bound.left_bound,
350                current_bound.right_bound,
351                mid,
352            );
353            if left_bound <= current_bound.right_bound {
354                queue.push_back(Bound {
355                    left_bound,
356                    right_bound: current_bound.right_bound,
357                    last_split_left_index: Some(mid),
358                    last_split_right_index: current_bound.last_split_right_index,
359                });
360            }
361            if current_bound.left_bound <= right_bound {
362                queue.push_back(Bound {
363                    left_bound: current_bound.left_bound,
364                    right_bound,
365                    last_split_left_index: current_bound.last_split_left_index,
366                    last_split_right_index: Some(mid),
367                });
368            }
369        }
370    }

372    #[allow(clippy::too_many_arguments)]
373    fn get_leaves_score(
374        &self,
375        view: &DataView<'_>,
376        feature: usize,
377        split_point: usize,
378        threshold_value: f64,
379        left_tree: &mut Tree,
380        right_tree: &mut Tree,
381        best_error: Cost,
382        min_sup: usize,
383        costs: &mut Depth2Loss<'_>,
384    ) {
385        let split_feature = view.get_sorted_feature(feature);
386        let feature_column_ids = view.get_feature_indices(feature);
387        let mut split_feature_split_indices = vec![0; view.total_instances];
388        let mut split_index: Option<usize> = None;
389
390        let size = feature_column_ids.len();
391        for i in 0..size {
392            let data_point = &split_feature[feature_column_ids[i]];
393            split_feature_split_indices[data_point.tid()] = data_point.unique_value_id();
394            if split_index.is_none() && data_point.value() > threshold_value {
395                split_index = Some(data_point.unique_value_id());
396            }
397        }
398
399        // The threshold lies between two values of the feature, so at least
400        // one value is above it.
401        let split_index = split_index.expect("a value lies above the threshold");
402
403        let dataset_size = view.get_dataset_size();
404        let class_number = view.get_num_labels();
405        debug_assert!(
406            split_point > 0 && split_point < dataset_size,
407            "left and right subtree need to be non-empty"
408        );
409
410        let mut left_leaves = SubtreeLeafScores::new(split_point, class_number, min_sup);
411        let mut right_leaves =
412            SubtreeLeafScores::new(dataset_size - split_point, class_number, min_sup);
413
414        view.initialize_split_parameters(
415            feature,
416            split_point,
417            &mut left_leaves.label_frequency,
418            &mut right_leaves.label_frequency,
419        );
420        let mut upper_bound = best_error;
421
422        for label in 0..class_number {
423            if left_leaves.label_frequency[label] > left_leaves.max_label_frequency {
424                left_leaves.max_label_frequency = left_leaves.label_frequency[label];
425                left_leaves.max_label = label;
426            }
427            if right_leaves.label_frequency[label] > right_leaves.max_label_frequency {
428                right_leaves.max_label_frequency = right_leaves.label_frequency[label];
429                right_leaves.max_label = label;
430            }
431        }
432        left_leaves.classification_score = left_leaves
433            .classification_score
434            .max(left_leaves.max_label_frequency);
435        right_leaves.classification_score = right_leaves
436            .classification_score
437            .max(right_leaves.max_label_frequency);
438        for leaves in [&mut left_leaves, &mut right_leaves] {
439            if costs.is_custom() {
440                (leaves.leaf_cost, leaves.leaf_label) = costs.leaf(&leaves.label_frequency);
441                leaves.best_cost = leaves.leaf_cost;
442            } else {
443                leaves.leaf_cost = classification_error(&leaves.label_frequency).0;
444                leaves.leaf_label = leaves.max_label;
445            }
446        }
447
448        for current_feature_index in 0..view.get_feature_number() {
449            if current_feature_index == feature {
450                self.process_depth_one_feature::<true>(
451                    view,
452                    feature,
453                    split_point,
454                    current_feature_index,
455                    split_index,
456                    &mut left_leaves,
457                    &mut right_leaves,
458                    &split_feature_split_indices,
459                    &mut upper_bound,
460                    costs,
461                );
462            } else {
463                self.process_depth_one_feature::<false>(
464                    view,
465                    feature,
466                    split_point,
467                    current_feature_index,
468                    split_index,
469                    &mut left_leaves,
470                    &mut right_leaves,
471                    &split_feature_split_indices,
472                    &mut upper_bound,
473                    costs,
474                );
475            }
476            let perfect = if costs.is_custom() {
477                left_leaves.best_cost == 0.0 && right_leaves.best_cost == 0.0
478            } else {
479                left_leaves.classification_score >= left_leaves.size
480                    && right_leaves.classification_score >= right_leaves.size
481            };
482            if perfect {
483                break;
484            }
485        }
486
487        if left_leaves.best_left_label.is_none() {
488            left_tree.update_root().map(|updater| {
489                updater
490                    .label(left_leaves.leaf_label)
491                    .error(left_leaves.leaf_cost)
492                    .leaf()
493            });
494        } else {
495            left_tree.update_root().map(|updater| {
496                updater
497                    .feature(left_leaves.best_feature_index)
498                    .split(left_leaves.best_threshold)
499                    .error(left_leaves.best_left_error + left_leaves.best_right_error)
500            });
501
502            let (ll, lr) = left_tree.node_children(left_tree.get_root_index());
503
504            left_tree.update_node(ll).map(|updater| {
505                updater
506                    .label(left_leaves.best_left_label.expect(LABELS_SET))
507                    .error(left_leaves.best_left_error)
508                    .leaf()
509            });
510
511            left_tree.update_node(lr).map(|updater| {
512                updater
513                    .label(left_leaves.best_right_label.expect(LABELS_SET))
514                    .error(left_leaves.best_right_error)
515                    .leaf()
516            });
517        }
518
519        if right_leaves.best_left_label.is_none() {
520            right_tree.update_root().map(|updater| {
521                updater
522                    .label(right_leaves.leaf_label)
523                    .error(right_leaves.leaf_cost)
524                    .leaf()
525            });
526        } else {
527            right_tree.update_root().map(|updater| {
528                updater
529                    .feature(right_leaves.best_feature_index)
530                    .split(right_leaves.best_threshold)
531                    .error(right_leaves.best_left_error + right_leaves.best_right_error)
532            });
533
534            let (ll, lr) = right_tree.node_children(right_tree.get_root_index());
535
536            right_tree.update_node(ll).map(|updater| {
537                updater
538                    .label(right_leaves.best_left_label.expect(LABELS_SET))
539                    .error(right_leaves.best_left_error)
540                    .leaf()
541            });
542
543            right_tree.update_node(lr).map(|updater| {
544                updater
545                    .label(right_leaves.best_right_label.expect(LABELS_SET))
546                    .error(right_leaves.best_right_error)
547                    .leaf()
548            });
549        }
550    }

552    #[allow(clippy::too_many_arguments)]
553    fn process_depth_one_feature<const IS_SAME_FEATURE: bool>(
554        &self,
555        view: &DataView<'_>,
556        _feature: usize,
557        split_point: usize,
558        current_feature_index: usize,
559        split_index: usize,
560        left_tree: &mut SubtreeLeafScores,
561        right_tree: &mut SubtreeLeafScores,
562        split_feature_split_indices: &[usize],
563        upper_bound: &mut Cost,
564        costs: &mut Depth2Loss<'_>,
565    ) {
566        let current_feature = view.get_sorted_feature(current_feature_index);
567        let feature_column_ids = view.get_feature_indices(current_feature_index);
568        let class_number = view.get_num_labels();
569        let total_size = left_tree.size + right_tree.size;
570
571        left_tree.reset_label_frequency();
572        right_tree.reset_label_frequency();
573
574        let mut index = 0;
575
576        for i in 0..feature_column_ids.len() {
577            let current_feature_data = &current_feature[feature_column_ids[i]];
578
579            if IS_SAME_FEATURE {
580                // Same feature as the root split: position decides the side.
581                if index < split_point {
582                    Self::process_single_point(
583                        current_feature_data,
584                        left_tree,
585                        class_number,
586                        current_feature_index,
587                        costs,
588                    );
589                } else {
590                    Self::process_single_point(
591                        current_feature_data,
592                        right_tree,
593                        class_number,
594                        current_feature_index,
595                        costs,
596                    );
597                }
598                index += 1;
599            } else {
600                // Other feature: look up the side from the root split's index.
601                let cur_split_index = split_feature_split_indices[current_feature_data.tid()];
602                if cur_split_index < split_index {
603                    Self::process_single_point(
604                        current_feature_data,
605                        left_tree,
606                        class_number,
607                        current_feature_index,
608                        costs,
609                    );
610                } else {
611                    Self::process_single_point(
612                        current_feature_data,
613                        right_tree,
614                        class_number,
615                        current_feature_index,
616                        costs,
617                    );
618                }
619            }
620
621            if !costs.is_custom() {
622                *upper_bound = (*upper_bound).min(total_size.saturating_sub(
623                    left_tree.classification_score + right_tree.classification_score,
624                ) as Cost);
625            }
626
627            if left_tree.is_zero && right_tree.is_zero {
628                break;
629            }
630
631            if left_tree.cannot_improve() && right_tree.cannot_improve() {
632                break;
633            }
634        }
635    }

637    /// Advances the scan of one side by one instance, first evaluating the
638    /// threshold just before it if its value differs from the previous one.
639    fn process_single_point(
640        current_feature_data: &DataPoint,
641        tree: &mut SubtreeLeafScores,
642        class_number: usize,
643        current_feature_index: usize,
644        costs: &mut Depth2Loss<'_>,
645    ) {
646        if costs.is_custom() {
647            Self::process_single_point_custom(
648                current_feature_data,
649                tree,
650                current_feature_index,
651                costs,
652            );
653            return;
654        }
655        if tree.is_zero {
656            return;
657        }
658
659        // The instance scanned last is now left of this threshold.
660        tree.can_skip = (tree.can_skip - 1.0).max(0.0);
661
662        // No threshold between equal values, or inside a skipped stretch.
663        if Some(current_feature_data.unique_value_id()) == tree.previous_unique_value_index
664            || tree.can_skip > 0.0
665        {
666            tree.move_left(current_feature_data);
667            return;
668        }
669
670        let mut left_classification_score = 0;
671        let mut right_classification_score = 0;
672        let mut left_label = None;
673        let mut right_label = None;
674
675        for label_value in 0..class_number {
676            let current_freq = tree.current_label_frequency[label_value];
677            if left_label.is_none() || current_freq > left_classification_score {
678                left_classification_score = current_freq;
679                left_label = Some(label_value);
680            }
681
682            let remaining =
683                tree.label_frequency[label_value] - tree.current_label_frequency[label_value];
684            if right_label.is_none() || remaining > right_classification_score {
685                right_classification_score = remaining;
686                right_label = Some(label_value);
687            }
688        }
689        let left_label = left_label.unwrap_or(0);
690        let right_label = right_label.unwrap_or(0);
691
692        let total_score = left_classification_score + right_classification_score;
693
694        let left_size = tree.current_element_count;
695        let right_size = tree.remaining();
696        let has_support = left_size >= tree.min_sup && right_size >= tree.min_sup;
697
698        if has_support && total_score > 0 && total_score > tree.classification_score {
699            tree.classification_score = total_score;
700            tree.best_feature_index = current_feature_index;
701            tree.best_threshold =
702                threshold_between(tree.previous_value, current_feature_data.value());
703
704            tree.best_left_label = Some(left_label);
705            tree.best_right_label = Some(right_label);
706            tree.best_left_error = tree.other_classes(left_label, true);
707            tree.best_right_error = tree.other_classes(right_label, false);
708        } else if costs.skipping {
709            tree.can_skip = tree.classification_score.saturating_sub(total_score) as Cost;
710        }
711
712        if costs.skipping {
713            // A pure right side only loses score as the threshold moves right,
714            // so no later threshold beats this one. That ends the scan only if
715            // this one could be chosen: while the left side is below the
716            // minimum support, a later threshold that meets it can still beat
717            // the best.
718            let right_is_pure = right_classification_score == right_size;
719            tree.is_zero |= (right_is_pure && left_size >= tree.min_sup) || tree.cannot_improve();
720        }
721
722        tree.move_left(current_feature_data);
723    }
724
725    /// `process_single_point` for a custom loss, which compares costs.
726    ///
727    /// Skipping and the early stops hold under the assumption `Skipping`
728    /// states, so they apply only when skipping is on. Then each evaluated
729    /// threshold is also checked against that assumption: since the one
730    /// evaluated before it, the left leaf gained, and the right leaf lost,
731    /// between 0 and the number of instances moved across.
732    fn process_single_point_custom(
733        current_feature_data: &DataPoint,
734        tree: &mut SubtreeLeafScores,
735        current_feature_index: usize,
736        costs: &mut Depth2Loss<'_>,
737    ) {
738        if tree.is_zero {
739            return;
740        }
741        tree.can_skip = (tree.can_skip - 1.0).max(0.0);
742        if Some(current_feature_data.unique_value_id()) == tree.previous_unique_value_index
743            || tree.can_skip > 0.0
744        {
745            tree.move_left(current_feature_data);
746            return;
747        }
748
749        for label in 0..tree.label_frequency.len() {
750            tree.remaining_frequency[label] =
751                tree.label_frequency[label] - tree.current_label_frequency[label];
752        }
753        let (left_cost, left_label) = costs.leaf(&tree.current_label_frequency);
754        let (right_cost, right_label) = costs.leaf(&tree.remaining_frequency);
755
756        if costs.skipping {
757            if let Some((previous_left, previous_right, previous_count)) = tree.last_evaluated {
758                let moved = (tree.current_element_count - previous_count) as Cost;
759                let within = |change: Cost| (0.0..=moved).contains(&change);
760                if !(within(left_cost - previous_left) && within(previous_right - right_cost)) {
761                    costs.violations += 1;
762                }
763            }
764            tree.last_evaluated = Some((left_cost, right_cost, tree.current_element_count));
765        }
766
767        let total_cost = left_cost + right_cost;
768        let left_size = tree.current_element_count;
769        let right_size = tree.remaining();
770        let has_support = left_size >= tree.min_sup && right_size >= tree.min_sup;
771
772        if has_support && total_cost < tree.best_cost {
773            tree.best_cost = total_cost;
774            tree.best_feature_index = current_feature_index;
775            tree.best_threshold =
776                threshold_between(tree.previous_value, current_feature_data.value());
777            tree.best_left_label = Some(left_label);
778            tree.best_right_label = Some(right_label);
779            tree.best_left_error = left_cost;
780            tree.best_right_error = right_cost;
781        } else if costs.skipping {
782            tree.can_skip = (total_cost - tree.best_cost).max(0.0);
783        }
784
785        if costs.skipping {
786            tree.is_zero |=
787                (right_cost == 0.0 && left_size >= tree.min_sup) || tree.cannot_improve();
788        }
789
790        tree.move_left(current_feature_data);
791    }
792}

794#[cfg(test)]
795mod d2_test {
796    use crate::algorithms::depth2::{ConTreeDepth2, Depth2Loss};
797    use crate::caching::Entry;
798    use crate::common::{
799        classification_error, Cost, Loss, PointSelector, SearchConfig, Statistics,
800    };
801    use crate::data::view::DataView;
802    use crate::reader::data_reader::DataReader;
803    use crate::reader::DataReaderError;
804
805    use crate::tests::fixture;
806
807    #[test]
808    fn depth_two_beats_the_majority_class_leaf() -> Result<(), DataReaderError> {
809        let reader = DataReader::default();
810        let mut dataset = reader.read_file(&fixture("hepatitis.txt"))?;
811        dataset.sort_features();
812
813        let root_view = DataView::root(&dataset, true);
814        let (leaf_error, leaf_label) = classification_error(root_view.get_labels_freqs());
815
816        let d2 = ConTreeDepth2;
817        // The majority-class leaf is the incumbent to beat.
818        let mut entry = Entry {
819            error: leaf_error,
820            label: leaf_label,
821            ..Entry::default()
822        };
823        let config = SearchConfig {
824            max_depth: 2,
825            min_sup: 1,
826            max_time: 10.0,
827            max_gap: 0.0,
828            max_error: Cost::INFINITY,
829            cost_step: 1.0,
830            skipping: crate::common::Skipping::Auto,
831            is_root: true,
832            use_heuristic: true,
833            fast_d2: true,
834            point_selector: PointSelector::default(),
835            nb_runs: 0,
836            discrepancy: 0,
837            budget: 0,
838        };
839
840        // No upper bound from a parent: this is the root.
841        let loss = Loss::default();
842        let mut invalid = None;
843        let tree = d2.fit(
844            &root_view,
845            &config,
846            &mut entry,
847            Cost::INFINITY,
848            &mut Statistics::default(),
849            &mut Depth2Loss::new(&loss, true, &mut invalid),
850        );
851
852        assert!(
853            entry.error <= leaf_error,
854            "a depth-2 tree scored {} where a single leaf scores {leaf_error}",
855            entry.error
856        );
857        assert!(!tree.is_empty(), "the specialized solver returned no tree");
858        assert_eq!(tree.root_error(), entry.error);
859
860        Ok(())
861    }
