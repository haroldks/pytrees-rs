13/// The threshold between two consecutive distinct values, `lower < upper`.
14///
15/// Rows go left when `x <= threshold`, so it has to hold that
16/// `lower <= threshold < upper`. The midpoint does, except that it can round
17/// up to `upper` when the two are adjacent doubles, in which case `lower` is
18/// used, as in scikit-learn.
19pub(crate) fn threshold_between(lower: f64, upper: f64) -> f64 {
20    let midpoint = (lower + upper) / 2.0;
21    if midpoint < upper {
22        midpoint
23    } else {
24        lower
25    }
26}

28/// The `(cost, label)` of a leaf under `loss`. A negative or NaN cost is kept
29/// in `invalid`, the first one only, for the fit to report, and replaced by
30/// an infinite cost so that the search never picks such a leaf.
31pub(crate) fn checked_leaf(
32    loss: &Loss,
33    class_counts: &[usize],
34    indices: impl FnOnce() -> Vec<usize>,
35    invalid: &mut Option<Cost>,
36) -> (Cost, usize) {
37    let (cost, label) = loss.leaf(class_counts, indices);
38    if cost.is_nan() || cost < 0.0 {
39        invalid.get_or_insert(cost);
40        return (Cost::INFINITY, label);
41    }
42    (cost, label)
43}

45/// Picks the candidate to evaluate inside an interval.
46pub(crate) fn select_point(selector: PointSelector, rng: &mut StdRng, bound: &Bound) -> usize {
47    match selector {
48        PointSelector::Mid => (bound.left_bound + bound.right_bound) / 2,
49        PointSelector::First => bound.left_bound,
50        PointSelector::Random => rng.random_range(bound.left_bound..=bound.right_bound),
51    }
52}

62/// Converts a cache entry into a tree node.
63///
64/// `Entry` marks a leaf with `feature == usize::MAX` and `split == INFINITY`;
65/// the tree marks one with `feature: None`.
66fn node_from_entry(entry: &Entry) -> NodeInfos {
67    let is_leaf = entry.feature == usize::MAX || !entry.split.is_finite();
68    NodeInfos {
69        feature: (!is_leaf).then_some(entry.feature),
70        split: (!is_leaf).then_some(entry.split),
71        error: entry.error,
72        label: Some(entry.label),
73    }
74}

76/// Reconstructs the solution tree from the cache, in canonical leaf form.
77///
78/// Empty when the cache holds no root, i.e. when `fit` has not run.
79pub(crate) fn build_solution_tree(cache: &Cache) -> Tree {
80    let mut solution = Tree::new();
81    if let Some(root) = cache.root() {
82        if let Some(tree_idx) = root.tree_idx {
83            if let Some(tree) = cache.get_tree(tree_idx) {
84                solution = tree.clone();
85            }
86        } else {
87            let infos = node_from_entry(root);
88            let root_index = solution.add_root(TreeNode::new(infos));
89            expand_children(cache, &mut solution, root_index, cache.root_index());
90        }
91    }
92    solution.normalize_leaves();
93    solution
94}

96fn expand_children(cache: &Cache, solution: &mut Tree, parent: usize, cache_index: usize) {
97    let branches = cache.get_children(cache_index);
98    for (branch, &child_index) in branches.iter().enumerate() {
99        // A cache index of 0 means "no child": index 0 is the root.
100        if child_index == 0 {
101            continue;
102        }
103        let Some(entry) = cache.get(child_index) else {
104            continue;
105        };
106
107        match entry.tree_idx {
108            // The depth-2 solver stores a whole subtree rather than cache
109            // entries, so it is grafted in one piece.
110            Some(tree_idx) => {
111                if let Some(sub_tree) = cache.get_tree(tree_idx) {
112                    let infos = sub_tree.root_details();
113                    let child = solution.add_node(parent, branch == 0, TreeNode::new(infos));
114                    solution.update_subtree(child, sub_tree, sub_tree.get_root_index());
115                }
116            }
117            None => {
118                let infos = node_from_entry(entry);
119                let child = solution.add_node(parent, branch == 0, TreeNode::new(infos));
120                if !entry.is_leaf {
121                    expand_children(cache, solution, child, child_index);
122                }
123            }
124        }
125    }
126}

128#[cfg(test)]
129mod tests {
130    use super::threshold_between;
131
132    #[test]
133    fn the_threshold_is_the_midpoint() {
134        assert_eq!(threshold_between(1.0, 2.0), 1.5);
135        assert_eq!(threshold_between(-3.0, 5.0), 1.0);
136    }
137
138    #[test]
139    fn between_adjacent_doubles_the_threshold_is_the_lower_one() {
140        // The midpoint of two adjacent doubles rounds to the one with an even
141        // mantissa. Here that is `upper`, and `x <= threshold` would then
142        // send `upper` left, against the split the search chose.
143        let lower = f64::from_bits(1.0_f64.to_bits() + 1);
144        let upper = f64::from_bits(lower.to_bits() + 1);
145        assert_eq!((lower + upper) / 2.0, upper);
146        assert_eq!(threshold_between(lower, upper), lower);
147    }
