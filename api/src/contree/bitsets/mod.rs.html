1//! Fixed-capacity bitsets over instance ids.
2//!
3//! Every node of the search is identified by the set of training instances
4//! that reach it, so these are hashed and compared constantly. Both the hash
5//! and the population count are memoised -- see [`Bitset::get_hash`] and
6//! [`Bitset::saved_count`] -- because recomputing either means another pass
7//! over `n / 64` words.
8
9use std::hash::{Hash, Hasher};
10use std::ops::Index;
11
12pub enum BitsetInit {
13    Empty(usize),
14    Full(usize),
15}
16
17#[derive(Debug, Clone)]
18pub struct Bitset {
19    capacity: usize,
20    count: usize,
21    hash: Option<u64>, // None = not computed yet
22    words: Vec<u64>,
23}
24
25pub trait BitCollection {
26    fn new(init: BitsetInit) -> Self;
27
28    fn count(&self) -> usize;
29
30    fn contains(&self, index: usize) -> bool;
31
32    fn set(&mut self, index: usize);
33
34    fn intersect_with(&self, other: &Bitset, invert: bool) -> Bitset;
35}
36
37impl BitCollection for Bitset {
38    fn new(init: BitsetInit) -> Self {
39        match init {
40            BitsetInit::Empty(n) => {
41                let word_count = n.div_ceil(64);
42                Self {
43                    capacity: n,
44                    count: 0,
45                    hash: None,
46                    words: vec![0u64; word_count],
47                }
48            }
49            BitsetInit::Full(n) => {
50                let word_count = n.div_ceil(64);
51                let mut words = vec![u64::MAX; word_count];
52                if n > 0 && n % 64 != 0 {
53                    if let Some(last) = words.last_mut() {
54                        *last = (1u64 << (n % 64)) - 1;
55                    }
56                }
57                Self {
58                    capacity: n,
59                    count: n,
60                    hash: None,
61                    words,
62                }
63            }
64        }
65    }
66
67    fn count(&self) -> usize {
68        self.words
69            .iter()
70            .map(|&word| word.count_ones() as usize)
71            .sum()
72    }
73
74    fn contains(&self, index: usize) -> bool {
75        debug_assert!(index < self.capacity, "Index out of bounds");
76        (self.words[index / 64] & (1u64 << (index % 64))) != 0
77    }
78
79    fn set(&mut self, index: usize) {
80        debug_assert!(index < self.capacity, "Index out of bounds");
81        self.words[index / 64] |= 1u64 << (index % 64);
82        self.hash = None; // Invalidate hash
83    }
84
85    fn intersect_with(&self, other: &Bitset, invert: bool) -> Bitset {
86        debug_assert_eq!(
87            self.capacity, other.capacity,
88            "Bitsets must have the same capacity"
89        );
90        let mut out = Bitset::new(BitsetInit::Empty(self.capacity));
91        for (i, (&word, &other_word)) in self.words.iter().zip(&other.words).enumerate() {
92            out.words[i] = word & if invert { !other_word } else { other_word };
93        }
94        out
95    }
96}
97
98impl Bitset {
99    /// Whether the hash has been computed.
100    pub fn is_hash_set(&self) -> bool {
101        self.hash.is_some()
102    }
103
104    /// The cached hash, computed on first use.
105    pub fn get_hash(&mut self) -> u64 {
106        match self.hash {
107            Some(hash) => hash,
108            None => self.compute_hash(),
109        }
110    }
111
112    /// Computes and caches the hash.
113    pub fn compute_hash(&mut self) -> u64 {
114        let mut h = self.words.len() as u64;
115        for &item in self.words.iter() {
116            h ^= item
117                .wrapping_add(0x9e3779b9)
118                .wrapping_add(h << 6)
119                .wrapping_add(h >> 2);
120        }
121        self.hash = Some(h);
122        h
123    }
124
125    /// Records the current population count for [`Self::saved_count`]. Call
126    /// it after the last mutation.
127    pub fn save_count(&mut self) {
128        self.count = self.count();
129    }
130
131    /// The population count recorded by the last [`Self::save_count`].
132    ///
133    /// Unlike `count()`, this does not scan the words.
134    pub fn saved_count(&self) -> usize {
135        debug_assert_eq!(
136            self.count,
137            self.count(),
138            "saved_count is stale: save_count was not called after the last mutation"
139        );
140        self.count
141    }
142}
143
144impl Index<usize> for Bitset {
145    type Output = u64;
146
147    fn index(&self, index: usize) -> &Self::Output {
148        debug_assert!(
149            index < self.words.len(),
150            "Index out for number of words bounds"
151        );
152        &self.words[index]
153    }
154}

156impl Hash for Bitset {
157    fn hash<H: Hasher>(&self, state: &mut H) {
158        if let Some(h) = self.hash {
159            h.hash(state);
160        } else {
161            let mut h = self.words.len() as u64;
162            for &item in self.words.iter() {
163                h ^= item
164                    .wrapping_add(0x9e3779b9)
165                    .wrapping_add(h << 6)
166                    .wrapping_add(h >> 2);
167            }
168            h.hash(state);
169        }
170    }
171}

173impl PartialEq for Bitset {
174    fn eq(&self, other: &Self) -> bool {
175        if let (Some(h1), Some(h2)) = (self.hash, other.hash) {
176            if h1 != h2 {
177                return false;
178            }
179        }
180
181        self.count == other.count && self.words == other.words
182    }
183}
